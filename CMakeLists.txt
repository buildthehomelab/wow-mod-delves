cmake_minimum_required(VERSION 3.16)
if(DEFINED AC_PATH_ROOT)
  return() # AzerothCore collects src/ and the mod_delves loader itself.
endif()
project(EvermoreDelvesFoundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(delve_run STATIC src/DelveRun.cpp)
target_include_directories(delve_run PUBLIC src)
target_compile_options(delve_run PRIVATE -Wall -Wextra -Werror -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(delve_run_tests tests/DelveRunTests.cpp)
  target_link_libraries(delve_run_tests PRIVATE delve_run)
  add_test(NAME delve_run_tests COMMAND delve_run_tests)
endif()
if(BUILD_TESTING)
  add_executable(delve_reward_tests tests/DelveRewardTests.cpp)
  target_link_libraries(delve_reward_tests PRIVATE delve_run)
  add_test(NAME delve_reward_tests COMMAND delve_reward_tests)
endif()
if(BUILD_TESTING)
  add_executable(delve_cart_tests tests/DelveCartTests.cpp)
  target_include_directories(delve_cart_tests PRIVATE src)
  target_compile_options(delve_cart_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
  add_test(NAME delve_cart_tests COMMAND delve_cart_tests)
endif()
